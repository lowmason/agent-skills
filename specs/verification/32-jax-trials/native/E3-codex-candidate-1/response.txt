For rotation equivariance, declare the group and every input/output representation first. For proper rotations, `R.T @ R = I` and `det(R) = +1`, the requirement is

\[
f(\rho_{\mathrm{in}}(R)x)=\rho_{\mathrm{out}}(R)f(x).
\]

With row-wise Cartesian coordinates, rotate positions and polar vector features by `x @ R.T`; scalar outputs stay unchanged, Cartesian vector outputs rotate, and rank-two Cartesian tensors transform as `R @ T @ R.T`. Higher-order irreducible channels need their declared representation matrices, rather than treating every feature as a Cartesian vector. O(3) additionally requires reflection/inversion parity; SO(3) alone makes no reflection claim. SE(3) additionally requires translation behavior. The e3nn-jax representation interface explicitly distinguishes degree and parity. [e3nn-jax irreps](https://e3nn-jax.readthedocs.io/en/latest/api/irreps.html).

```python
from dataclasses import dataclass

import diffrax as dfx
import jax
import jax.numpy as jnp
import numpy as np
from jax.flatten_util import ravel_pytree

jax.config.update("jax_enable_x64", True)


@dataclass(frozen=True)
class SolvePolicy:
    name: str
    solver: object
    rtol: float
    atol: float
    dtype: str = "float64"
    dt0: float = 0.01
    max_steps: int = 100_000


def solve(vf, params, y0, times, policy, *, throw=False):
    dtype = jnp.dtype(policy.dtype)
    if dtype == jnp.float64 and not jax.config.x64_enabled:
        raise ValueError("float64 policy requires enabled x64")
    # params here is a floating-array pytree; partition non-array model metadata.
    params = jax.tree_util.tree_map(lambda a: jnp.asarray(a, dtype), params)
    y0, times = jnp.asarray(y0, dtype), jnp.asarray(times, dtype)
    return dfx.diffeqsolve(
        dfx.ODETerm(vf), policy.solver,
        t0=times[0], t1=times[-1], dt0=policy.dt0,
        y0=y0, args=params, saveat=dfx.SaveAt(ts=times),
        stepsize_controller=dfx.PIDController(
            rtol=policy.rtol, atol=policy.atol
        ),
        adjoint=dfx.RecursiveCheckpointAdjoint(),
        max_steps=policy.max_steps, throw=throw,
    )


def discrepancy(actual, expected, *, atol, rtol, scale_floor):
    actual, expected = np.asarray(actual), np.asarray(expected)
    if actual.shape != expected.shape or actual.size == 0:
        raise ValueError("Need nonempty aligned arrays")
    finite = np.isfinite(actual).all() and np.isfinite(expected).all()
    if not finite:
        return {"finite": False, "passed": False,
                "max_abs": None, "normalized_rms": None,
                "max_scaled": None}
    delta = actual - expected
    rms = lambda a: float(np.sqrt(np.mean(np.square(a))))
    scale = max(rms(actual), rms(expected), scale_floor)
    threshold = atol + rtol * np.maximum(np.abs(actual), np.abs(expected))
    scaled = float(np.max(np.abs(delta) / threshold))
    return {"finite": True, "passed": scaled <= 1.0,
            "max_abs": float(np.max(np.abs(delta))),
            "normalized_rms": rms(delta) / scale,
            "max_scaled": scaled}


def ode_sweep(vf, params, ids, initial, times, targets, valid, reference,
              policies, *, numerical_atol, numerical_rtol, scale_floor):
    initial, times = np.asarray(initial), np.asarray(times)
    targets, reference, valid = map(np.asarray, (targets, reference, valid))
    if (targets.ndim != 3 or reference.shape != targets.shape
            or valid.shape != targets.shape or valid.dtype != np.bool_
            or initial.shape != (targets.shape[0], targets.shape[2])
            or len(ids) != targets.shape[0] or len(set(ids)) != len(ids)
            or times.shape != (targets.shape[1],) or len(times) < 2
            or not np.all(np.diff(times) > 0)):
        raise ValueError("Invalid trajectory/time/channel contract")
    if (not np.isfinite(initial).all() or not np.isfinite(times).all()
            or not valid.reshape(len(ids), -1).any(axis=1).all()
            or not np.isfinite(targets[valid]).all()
            or not np.isfinite(reference[valid]).all()):
        raise ValueError("Invalid eligible inputs, targets, or reference")
    rows = []
    for policy in policies:
        for index, unit_id in enumerate(ids):
            sol = solve(vf, params, initial[index], times, policy)
            predictions = np.asarray(sol.ys)
            if predictions.shape != targets[index].shape:
                raise ValueError("Saved prediction shape mismatch")
            if predictions.dtype != np.dtype(policy.dtype):
                raise ValueError("Actual solve dtype differs from policy")
            mask = valid[index]
            success = bool(np.asarray(sol.result == dfx.RESULTS.successful))
            success = success and np.isfinite(predictions[mask]).all()
            row = {
                "unit_id": unit_id, "policy": policy.name,
                "dtype": str(predictions.dtype),
                "result": str(sol.result), "successful": bool(success),
                "stats": {k: int(np.asarray(v)) for k, v in sol.stats.items()
                          if k in {"num_steps", "num_accepted_steps",
                                   "num_rejected_steps"}},
                "eligible_count": int(mask.sum()),
                "predictions": predictions, "task_mse": None,
                "numerical_discrepancy": None,
            }
            if success:
                row["task_mse"] = float(np.mean(np.square(
                    predictions[mask] - targets[index][mask]
                )))
                row["numerical_discrepancy"] = discrepancy(
                    predictions[mask], reference[index][mask],
                    atol=numerical_atol, rtol=numerical_rtol,
                    scale_floor=scale_floor,
                )
            rows.append(row)
    # Do not silently aggregate only successful units. Apply a declared failure
    # rule, and report failure counts with any aggregate task score.
    return rows


def gradient_directions(loss, params, directions, epsilons):
    flat, unravel = ravel_pytree(params)
    flat_loss = lambda vector: loss(unravel(vector))
    gradient = jax.grad(flat_loss)(flat)
    if not np.isfinite(np.asarray(gradient)).all():
        raise ValueError("Nonfinite autodiff gradient")
    rows = []
    for index, direction in enumerate(directions):
        direction = jnp.asarray(direction, dtype=flat.dtype)
        norm = float(jnp.linalg.norm(direction))
        if direction.shape != flat.shape or not np.isfinite(norm) or norm == 0:
            raise ValueError("Invalid parameter direction")
        direction = direction / norm
        autodiff = float(jnp.vdot(gradient, direction))
        for epsilon in epsilons:
            if epsilon <= 0:
                raise ValueError("Epsilon must be positive")
            forward = float(flat_loss(flat + epsilon * direction))
            backward = float(flat_loss(flat - epsilon * direction))
            finite_difference = (forward - backward) / (2 * epsilon)
            rows.append({
                "direction": index, "epsilon": float(epsilon),
                "autodiff": autodiff, "finite_difference": finite_difference,
                "finite": bool(np.isfinite(finite_difference)),
                "absolute_difference": abs(autodiff - finite_difference),
            })
    # Inspect the whole epsilon/tolerance table, not just its best-looking row.
    return np.asarray(gradient), rows


def proper_rotation(rng):
    matrix, _ = jnp.linalg.qr(jnp.asarray(rng.normal(size=(3, 3))))
    matrix = matrix.at[:, -1].multiply(jnp.linalg.det(matrix))
    rotation = np.asarray(matrix)
    np.testing.assert_allclose(rotation.T @ rotation, np.eye(3), atol=1e-12)
    np.testing.assert_allclose(np.linalg.det(rotation), 1.0, atol=1e-12)
    return matrix


def equivariance_rows(pipeline, examples, transformations,
                      input_action, output_action, *, atol, rtol, scale_floor):
    rows = []
    for unit_id, example in examples:
        original = pipeline(example)
        for name, transformation in transformations:
            actual = pipeline(input_action(example, transformation))
            expected = output_action(original, transformation)
            if set(actual) != set(expected):
                raise ValueError("Output channel mismatch")
            for channel in expected:
                rows.append({
                    "unit_id": unit_id, "transformation": name,
                    "channel": channel,
                    **discrepancy(actual[channel], expected[channel],
                                  atol=atol, rtol=rtol,
                                  scale_floor=scale_floor),
                })
    return rows


# Explicit fixture representations: invariant graph scalar, polar node vectors,
# and a graph Cartesian rank-two tensor. No reflection claim is tested here.
def point_input_action(example, transformation):
    rotation, permutation = transformation
    return {
        "positions": (example["positions"] @ rotation.T)[permutation],
        "weights": example["weights"][permutation],
    }


def point_output_action(output, transformation):
    rotation, permutation = transformation
    return {
        "scalar": output["scalar"],
        "vectors": (output["vectors"] @ rotation.T)[permutation],
        "tensor": rotation @ output["tensor"] @ rotation.T,
    }


def equivariant_control(example):
    centered = example["positions"] - example["positions"].mean(axis=0)
    weighted = centered * example["weights"][:, None]
    return {
        "scalar": jnp.sum(centered * weighted),
        "vectors": weighted,
        "tensor": centered.T @ weighted,
    }


def broken_control(example):
    output = equivariant_control(example)
    return {**output,
            "vectors": output["vectors"] @ jnp.diag(jnp.array([1., 2., 3.]))}


def main():
    # Analytic positive control: y' = theta*y, y(t) = y0*exp(theta*t).
    theta = jnp.array(0.7, dtype=jnp.float64)
    times = np.linspace(0., 1., 21)
    initial = np.array([[1.], [2.]])
    reference = initial[:, None, :] * np.exp(float(theta) * times[None, :, None])
    policies = [
        SolvePolicy("loose-f32", dfx.Tsit5(), 1e-3, 1e-5, "float32"),
        SolvePolicy("loose-f64", dfx.Tsit5(), 1e-3, 1e-5),
        SolvePolicy("medium-f64", dfx.Tsit5(), 1e-6, 1e-8),
        SolvePolicy("tight-f64", dfx.Tsit5(), 1e-9, 1e-11),
        SolvePolicy("tighter-f64", dfx.Tsit5(), 1e-11, 1e-13),
        SolvePolicy("independent-f64", dfx.Dopri8(), 1e-11, 1e-13),
    ]
    vf = lambda t, y, parameter: parameter * y
    ode_rows = ode_sweep(
        vf, theta, ("analytic-a", "analytic-b"), initial, times,
        reference, np.ones(reference.shape, dtype=bool), reference, policies,
        numerical_atol=1e-8, numerical_rtol=1e-8, scale_floor=1e-12,
    )
    # Audit the endpoint derivative against the analytic derivative. A real audit
    # should also use its training loss and its actual training adjoint.
    endpoint = lambda parameter: solve(
        vf, parameter, jnp.array([2.]), times, policies[-2], throw=True
    ).ys[-1, 0]
    derivative, gradient_rows = gradient_directions(
        endpoint, theta, (np.array([1.]),), np.logspace(-2, -7, 6)
    )
    analytic_derivative = 2 * np.exp(float(theta))
    print("analytic derivative:", analytic_derivative,
          "autodiff derivative:", derivative)
    print("gradient epsilon table:", gradient_rows)
    print("ODE table:", [{k: v for k, v in row.items() if k != "predictions"}
                         for row in ode_rows])

    rng = np.random.default_rng(2026)
    rotation_1, rotation_2 = proper_rotation(rng), proper_rotation(rng)
    permutation = jnp.array([2, 0, 3, 1])
    identity_permutation = jnp.arange(4)
    fixed_rotation = jnp.array([[0., -1., 0.], [1., 0., 0.], [0., 0., 1.]])
    transformations = [
        ("identity", (jnp.eye(3), identity_permutation)),
        ("fixed-z-rotation", (fixed_rotation, identity_permutation)),
        ("random-1", (rotation_1, identity_permutation)),
        ("random-2", (rotation_2, identity_permutation)),
        ("composition", (rotation_2 @ rotation_1, identity_permutation)),
        ("relabel", (jnp.eye(3), permutation)),
        ("rotation-and-relabel", (rotation_1, permutation)),
    ]
    examples = [("generic", {
        "positions": jnp.array([[1., 2., 0.], [-1., 0., 3.],
                                [2., -2., 1.], [0., 1., -1.]]),
        "weights": jnp.array([1., 2., 3., 4.]),
    })]
    good = equivariance_rows(
        equivariant_control, examples, transformations,
        point_input_action, point_output_action,
        atol=1e-10, rtol=1e-10, scale_floor=1e-12,
    )
    bad = equivariance_rows(
        broken_control, examples, transformations,
        point_input_action, point_output_action,
        atol=1e-10, rtol=1e-10, scale_floor=1e-12,
    )
    assert all(row["passed"] for row in good), good
    assert any(not row["passed"] for row in bad), bad
    print("positive rotation control:", good)
    print("negative rotation control:", bad)
    print("environment:", {"jax": jax.__version__, "diffrax": dfx.__version__,
                           "numpy": np.__version__, "backend": jax.default_backend(),
                           "devices": [str(device) for device in jax.devices()],
                           "x64_enabled": jax.config.x64_enabled})


if __name__ == "__main__":
    main()
```
