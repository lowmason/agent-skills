- [x] **Step 4 (A2): Write and run the log-prior repro against the CURRENT recipe (RED).** Write `<scratchpad>/repro_a2.py` — it embeds the hierarchical.md correlated-varying-effects model and both the current and fixed `add_log_prior` bodies:

```python
import numpy as np
import jax
import jax.numpy as jnp
import numpyro
import numpyro.distributions as dist
from numpyro.handlers import trace, substitute
from numpyro.infer import MCMC, NUTS
import arviz as az
import xarray as xr

numpyro.set_host_device_count(2)
rng_key = jax.random.PRNGKey(7)
n_groups, n_obs = 4, 80
rng = np.random.default_rng(7)
group_idx = np.repeat(np.arange(n_groups), n_obs // n_groups)
xdat = rng.normal(size=n_obs)
ydat = 1.0 + 0.5 * xdat + rng.normal(0, 0.3, size=n_obs)

def model(x, group_idx, y=None):
    L_omega = numpyro.sample('L_omega', dist.LKJCholesky(2, concentration=2.0))
    sd = numpyro.sample('sd', dist.Exponential(jnp.ones(2)))
    scale_tril = sd[..., None] * L_omega
    mu_i = numpyro.sample('mu_intercept', dist.Normal(0.0, 5.0))
    mu_s = numpyro.sample('mu_slope', dist.Normal(0.0, 1.0))
    loc = jnp.stack([mu_i, mu_s])
    with numpyro.plate('group', n_groups):
        effects = numpyro.sample('effects', dist.MultivariateNormal(loc, scale_tril=scale_tril))
    sigma = numpyro.sample('sigma', dist.HalfNormal(1.0))
    mu = effects[group_idx, 0] + effects[group_idx, 1] * x
    with numpyro.plate('obs', x.shape[0]):
        numpyro.sample('y_obs', dist.Normal(mu, sigma), obs=y)

mcmc = MCMC(NUTS(model), num_warmup=100, num_samples=100, num_chains=2, progress_bar=False)
mcmc.run(rng_key, xdat, group_idx, y=ydat)
idata = az.from_numpyro(mcmc)

samples = mcmc.get_samples(group_by_chain=True)
chains, draws = next(iter(samples.values())).shape[:2]
flat = {k: v.reshape((chains * draws,) + v.shape[2:]) for k, v in samples.items()}

def per_draw(params):
    tr = trace(substitute(model, params)).get_trace(xdat, group_idx, y=ydat)
    return {name: site['fn'].log_prob(site['value'])
            for name, site in tr.items()
            if site['type'] == 'sample' and not site.get('is_observed', False)}

lp = jax.vmap(per_draw)(flat)
post = idata['posterior'].dataset

# --- CURRENT recipe: must raise ---
try:
    data_vars = {
        name: (post[name].dims, np.asarray(v).reshape((chains, draws) + v.shape[1:]))
        for name, v in lp.items() if name in post
    }
    xr.Dataset(data_vars, coords={'chain': post.chain, 'draw': post.draw})
    raise SystemExit('FAIL: current recipe did not raise on event-dim sites')
except ValueError as exc:
    print(f'current recipe raises as documented: {type(exc).__name__}')

# --- FIXED recipe: must succeed with rank-matched dims ---
data_vars = {}
for name, v in lp.items():
    if name not in post:
        continue
    arr = np.asarray(v).reshape((chains, draws) + v.shape[1:])
    data_vars[name] = (post[name].dims[:arr.ndim], arr)
ds = xr.Dataset(data_vars, coords={'chain': post.chain, 'draw': post.draw})
assert ds['L_omega'].dims == ('chain', 'draw'), ds['L_omega'].dims
assert ds['effects'].dims == ('chain', 'draw', 'effects_dim_0'), ds['effects'].dims
print('A2 repro: PASS (current recipe raises; fixed recipe builds log_prior)')
```

```python
def _parse_frontmatter(md: Path) -> tuple[dict | None, list[str]]:
    text = md.read_text()
    m = re.match(r'^---\n(.*?)\n---\n', text, re.S)
    if not m:
        return None, [f'{md}: no frontmatter block']
    try:
        fm = yaml.safe_load(m.group(1))
    except yaml.YAMLError as exc:
        return None, [f'{md}: frontmatter is not valid YAML ({type(exc).__name__})']
    if not isinstance(fm, dict):
        return None, [f'{md}: frontmatter is not a mapping']
    return fm, []


def check_agent_file(md: Path) -> list[str]:
    fm, errs = _parse_frontmatter(md)
    if fm is None:
        return errs
    if fm.get('name') != md.stem:
        errs.append(f'{md}: name {fm.get("name")!r} does not match filename {md.stem!r}')
    if not (fm.get('description') or '').strip():
        errs.append(f'{md}: missing description')
    tools = fm.get('tools')
    if tools is not None:
        unknown = [t.strip() for t in str(tools).split(',') if t.strip() not in KNOWN_AGENT_TOOLS]
        if unknown:
            errs.append(f'{md}: unknown tools {unknown}')
    return errs


def check_command_file(md: Path) -> list[str]:
    fm, errs = _parse_frontmatter(md)
    if fm is None:
        return errs
    if not (fm.get('description') or '').strip():
        errs.append(f'{md}: missing description')
    return errs
```
