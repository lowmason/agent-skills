### R3 — The runner: `build/run_suites.py`
S79. R3.1 Stdlib only (`tomllib`, `argparse`, `subprocess`, `shlex`), Python 3.13.
S80. Invoked as `uv run --python 3.13 python build/run_suites.py …` from any directory.
S81. It finds the repo root from its own path, never from the working directory.
S82. R3.2 **One run** is one interpreter of one suite.
S83. Its command is `uv run --python <python> --with <dep> … python -m pytest -q [<args>]`, run with the suite's `dir` as its working directory.
S84. Its shell form is `cd <dir> && <command>`, with shell quoting where a word needs it.
S85. With default options, the shell form equals the suite's current `CLAUDE.md` command, character for character.
S86. For `hooks`, whose line chains two runs as `cd hooks && A && B`, the two shell forms are `cd hooks && A` and `cd hooks && B`.
S87. R3.3 **`--list [NAME …]`** prints, for each named suite (default: all): its name and directory, the shell form of each run, its extras, and its notes.
S88. `--extras` and the `--` arguments apply here too, so `--list` prints the exact command that a run would use.
S89. R3.4 **`NAME [NAME …]`** runs the named suites, in the order given, one run per interpreter in `pythons`.
S90. Each run starts with a `== <name> (<python>)` header, and the suite's notes follow its first run's header.
S91. Then pytest's output streams.
S92. A failed run does not stop the next one.
S93. R3.5 **`--extras`** adds each suite's `extras` after its `with` deps.
S94. Arguments after **`--`** go to pytest (a test node ID, `-k`, `-s`, `-rs`) in every run.
S95. R3.6 **`--all`** runs every suite except one whose `dir` lies inside another suite's `dir`.
S96. Today that is `cc-guide`: `cd build && pytest` collects `build/cc_guide` already.
S97. `--all` prints one line for each suite that it skips.
S98. A suite named on the command line always runs.
S99. R3.7 **Exit status:** 0 when every run exits 0; 1 when any run fails (pytest's exit 5, no tests collected, counts as a failure); 2 for a usage error, an unknown suite name, or a registry that fails validation.
S100. The last line names the failed runs, or says that every run passed.
S101. A run whose `uv` cannot start counts as failed.
S102. R3.8 The runner passes its environment to every run unchanged.
S103. `CALIBRATION_SWEEP=1 uv run --python 3.13 python build/run_suites.py bayesian-workflow -- -s` therefore runs the sweep.
S104. R3.9 Validation fails closed.
S105. An unknown key, a missing required key, a wrong type, a duplicate name or directory, or a `with` list without `pytest` stops `--list`, a named run and `--all` with exit 2.
S106. `--check` reports the same failures as findings (R4.3).
S107. A misspelled key can therefore never run a suite without its deps.

407| @dataclass(frozen=True)
408| class Suite:
409|     name: str
410|     dir: str
411|     deps: tuple[str, ...]  # the registry's `with` list
412|     extras: tuple[str, ...] = ()
413|     pythons: tuple[str, ...] = DEFAULT_PYTHONS
414|     notes: tuple[str, ...] = ()
415| 
416| 
417| @dataclass(frozen=True)
418| class Run:
419|     '''One interpreter's pytest run of one suite.'''
420|     suite: Suite
421|     python: str
422|     argv: tuple[str, ...]
423| 
424|     @property
425|     def label(self) -> str:
426|         return f'{self.suite.name} ({self.python})'
427| 
428|     def shell(self) -> str:
429|         '''The run as one shell line, in the form the root CLAUDE.md used.'''
430|         return f'cd {shlex.quote(self.suite.dir)} && {shlex.join(self.argv)}'
431| 
432| 
433| def _is_string_list(value: object) -> bool:
434|     return isinstance(value, list) and all(isinstance(v, str) and v for v in value)
435| 
436| 
437| def _dir_problem(value: str) -> str | None:
438|     path = PurePosixPath(value)
439|     if path.is_absolute() or '..' in path.parts or str(path) != value or value == '.':
440|         return "'dir' must be a normalised path below the repo root"
441|     return None
442| 
443| 
444| def validate(data: dict) -> list[str]:
445|     '''Every problem that keeps the registry from being used; empty when it is valid.'''
446|     problems = [f'unknown top-level key {key!r}' for key in data if key != 'suite']
447|     tables = data.get('suite')
448|     if not isinstance(tables, list) or not tables:
449|         return [*problems, 'no [[suite]] tables']
450|     names: set[str] = set()
451|     dirs: set[str] = set()
452|     for i, table in enumerate(tables, start=1):
453|         if not isinstance(table, dict):
454|             problems.append(f'suite {i}: not a table')
455|             continue
456|         name = table.get('name')
457|         where = f'suite {i} ({name})' if isinstance(name, str) and name else f'suite {i}'
458|         problems += [f'{where}: unknown key {key!r}' for key in table
459|                      if key not in REQUIRED_KEYS + OPTIONAL_KEYS]
460|         problems += [f'{where}: missing key {key!r}' for key in REQUIRED_KEYS if key not in table]
461|         problems += [f'{where}: {key!r} must be a non-empty string' for key in STRING_KEYS
462|                      if key in table and not (isinstance(table[key], str) and table[key])]
463|         problems += [f'{where}: {key!r} must be a list of non-empty strings' for key in LIST_KEYS
464|                      if key in table and not _is_string_list(table[key])]
465|         if _is_string_list(table.get('with')) and 'pytest' not in table['with']:
466|             problems.append(f"{where}: 'with' must include 'pytest'")
467|         if table.get('pythons') == []:
468|             problems.append(f"{where}: 'pythons' must not be empty")
469|         if isinstance(name, str) and name:
470|             if name in names:
471|                 problems.append(f'{where}: duplicate name')
472|             names.add(name)
473|         dir_ = table.get('dir')
474|         if isinstance(dir_, str) and dir_:
475|             if problem := _dir_problem(dir_):
476|                 problems.append(f'{where}: {problem}')
477|             if dir_ in dirs:
478|                 problems.append(f'{where}: duplicate dir {dir_!r}')
479|             dirs.add(dir_)
480|     return problems
481| 
482| 
483| def parse(data: dict) -> list[Suite]:
484|     '''The suites of a registry that validate() passes, in file order.'''
485|     return [Suite(name=t['name'], dir=t['dir'], deps=tuple(t['with']),
486|                   extras=tuple(t.get('extras', ())),
487|                   pythons=tuple(t.get('pythons', DEFAULT_PYTHONS)),
488|                   notes=tuple(t.get('notes', ())))
489|             for t in data['suite']]
490| 
491| 
492| def runs(suite: Suite, extras: bool = False, pytest_args: Sequence[str] = ()) -> list[Run]:
493|     '''One run per interpreter, in the suite's order; --extras adds the extras.'''
494|     deps = suite.deps + (suite.extras if extras else ())
495|     withs = tuple(word for dep in deps for word in ('--with', dep))
496|     return [Run(suite, python, ('uv', 'run', '--python', python, *withs,
497|                                 'python', '-m', 'pytest', '-q', *pytest_args))
498|             for python in suite.pythons]
499| 
500| 
501| def container(suite: Suite, suites: Sequence[Suite]) -> Suite | None:
502|     '''The other suite whose directory holds this suite's directory, if any.'''
503|     path = PurePosixPath(suite.dir)
504|     return next((other for other in suites
505|                  if other.dir != suite.dir and path.is_relative_to(other.dir)), None)
506| ```

925|     args = parser.parse_args(argv)

927|         parser.error('--check takes no suite names, --extras or pytest arguments')

929|         parser.error('--all takes no suite names')
930|     if not (args.list or args.all or args.check or args.names):
931|         parser.error('name a suite, or use --list, --all or --check')
932|     return args, pytest_args

935| def main(argv: Sequence[str] | None = None, root: Path = REPO,

940|     except (OSError, UnicodeDecodeError, tomllib.TOMLDecodeError) as exc:
941|         print(f'{REGISTRY}: cannot read: {exc}', file=sys.stderr)
942|         return 1 if args.check else 2

944|         problems = check(root, data)
945|         for problem in problems:
946|             print(problem)
947|         if problems:
948|             return 1
949|         print('suite registry ok')
950|         return 0
951|     problems = validate(data)
952|     if problems:
953|         for problem in problems:
954|             print(f'{REGISTRY}: {problem}', file=sys.stderr)
955|         return 2
956|     suites = parse(data)
957|     by_name = {suite.name: suite for suite in suites}
958|     unknown = [name for name in args.names if name not in by_name]
959|     if unknown:

961|         return 2
